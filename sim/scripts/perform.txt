# Performance mode. Track 1: a steady tone, left for the first second of a
# 2 s loop, right for the second, so the channel tells where the loop is read.
# Pads: track row, columns 5..12 = Rpt 1/8, Rpt 1/16, Low-pass, High-pass,
# Sweep dn, Sweep up, Vibrato, Tape stop (track 1 pads 4..11).
wait 300
tone_shape steady
tone 330
tone_ch left
tap 0
wait 1000
tone_ch right
wait 1000
tap 0
tone off
expect_len 1 2.0

# Tap Perform: the overlay stays on.
press PERFORMANCE
wait 50
shot ../build/sim/check/perform_overlay.png

# Beat repeat early in the loop: it keeps repeating the left half past the
# point where the loop moves to the right half...
down 4
wait 100
expect_perf 1 1 on
reset_levels
wait 1300
expect_ch left 0.2
expect_ch right silent
shot ../build/sim/check/perform_repeat.png
# ...and on release it lands back in time, in the right half.
up 4
wait 10
expect_perf 1 1 off
wait 50
reset_levels
wait 300
expect_ch right 0.2

# Track 2: a 4 kHz tone. Low-pass cut held: almost nothing left.
tone_shape steady
tone 4000
tone_ch both
tap 12
wait 2000
tap 12
# The take started mid-bar, so "On bar" records on to the next bar line.
wait 2100
expect_state 2 PLAY
tone off
tap 2
wait 500
reset_levels
wait 300
expect_audio 0.3
down 18
wait 100
reset_levels
wait 400
expect_below 0.05
up 18

# Track 3: a 100 Hz tone. High-pass cut held: almost nothing left.
tone 100
tap 24
wait 2000
tap 24
wait 2100
expect_state 3 PLAY
tone off
tap 14
wait 500
reset_levels
wait 300
expect_audio 0.3
down 31
wait 100
reset_levels
wait 400
expect_below 0.05
up 31
tap 26

# Tape stop on track 1 (the others muted): silent after two beats.
tap 2
wait 200
down 11
wait 1100
reset_levels
wait 300
expect_silent
up 11
wait 100
reset_levels
wait 300
expect_audio 0.2

# Vibrato still sounds.
down 10
wait 100
reset_levels
wait 400
expect_audio 0.2
up 10

# Shift + pad latches (Sweep up, effect 6); it stays on after the pad is up.
hold SHIFT
down 9
up 9
release SHIFT
wait 300
expect_perf 1 6 on
shot ../build/sim/check/perform_latched.png

# Tap Perform again: off, and everything with it.
press PERFORMANCE
wait 50
expect_perf 1 6 off

# Hold Perform (momentary): on while held, off at release.
hold PERFORMANCE
wait 500
down 6
wait 200
expect_perf 1 3 on
up 6
release PERFORMANCE
wait 50
expect_perf 1 3 off
shot ../build/sim/check/perform_closed.png
