# Track options: quantised stop, free stop, count-in with a fixed length, mono input.
# Tempo 120 bpm, 4 beats: a bar is 2.0 s until the first loop sets it.
wait 300
# These checks were written for takes that start at once (the default is a count-in).
opt all start 0

# Track 1, stop "On bar" (the default): stopping 0.5 s into bar 2 records on to its end.
tap 0
wait 2500
tap 0
expect_state 1 REC
wait 100
shot ../build/sim/check/options_pending_stop.png
wait 1500
expect_state 1 PLAY
expect_len 1 4.0

# Track 2, stop "Free": the take is exactly as long as played. Shift + F5
# taps cycle On bar -> On beat -> Free on the selected track.
press DOWN
hold SHIFT
press F5
wait 50
expect_opt 2 stop 2
press F5
wait 100
expect_opt 2 stop 0
shot ../build/sim/check/options_shift.png
release SHIFT
# Shift + F7 cycles After rec, Shift + F8 the tempo mode (back again after).
hold SHIFT
press F7
wait 30
expect_opt 2 after 1
press F7
press F7
press F8
wait 30
expect_opt 2 after 0
expect_opt 2 stretch 1
press F8
wait 30
expect_opt 2 stretch 0
release SHIFT
tap 12
wait 1300
tap 12
expect_len 2 1.32
