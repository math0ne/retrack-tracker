# Long takes: past ~19.5 s (its RAM less the 2 s ring) a take goes on to the
# track's stream file on the card, up to 5 minutes. A 30 s take: 330 Hz bursts
# for 15 s, then 660 Hz; the 660 Hz part from 19.5 s on plays from the card.
wait 300
opt all start 0
# On bar (the default): 15 bars of 2 s, so the bar (120 bpm) is the tempo's.
# A steady tone (the pitch probe counts over the whole window); the tap takes 20 ms.
tone_shape steady
tone 330
tap 0
wait 14980
tone 660
wait 15000
tap 0
tone off
wait 50
expect_stream 1 long 1
expect_len 1 30.0
expect_stream 1 over 0
shot ../build/sim/check/stream_long.png
