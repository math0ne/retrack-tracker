# Long takes on a card that takes time, like the 8 GB test card with 32 KB
# clusters and direct DMA: 8 MB/s writes, 16 MB/s reads, 1 ms a command, a
# 150 ms stall every 64th write. The main loop is blocked during each card
# command while the audio runs on, as on the device. Defaults: count-in, on-bar.
card_speed 8 16 1 150 64
wait 300
tone_shape steady
# Track 1: a long take, 330 Hz to 14 s, then 660 Hz (from the card after 19.5 s).
tone 330
tap 0
wait 16000
tone 660
wait 16000
tap 0
wait 3000
tone off
expect_state 1 PLAY
expect_stream 1 long 1
expect_stream 1 over 0
# Track 2: a short take, then Play twice (stop, everything from bar 1).
press DOWN

tap 12
wait 3000
tone off

expect_stream 2 long 0

press PLAY

press PLAY

reset_levels
wait 1000
expect_pitch 330
wait 20000

wait 1500

wait 12000
