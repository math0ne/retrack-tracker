# A free first take after a count-in sets the bar to its own length, and its
# loop starts on bar 1: the first bar line is its start, so the metronome,
# later count-ins and Play all line up with it.
wait 300
opt all stop 0
tone_shape steady
tone 330
tap 0
wait 5100
tap 0
wait 500
tone off
expect_state 1 PLAY
expect_baroffset 1 0

press PLAY
